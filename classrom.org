<script>

// SAMPLE COLLEGE DATA

const lectures = [
  {
    department:"VLSI",
    year:"First Year",
    division:"A",
    room:"204",
    time:"09:00 - 10:00",
    subject:"Engineering Mathematics",
    teacher:"Professor A",
    status:"Scheduled"
  },
  {
    department:"VLSI",
    year:"First Year",
    division:"A",
    room:"204",
    time:"10:00 - 11:00",
    subject:"Basic Electronics",
    teacher:"Professor B",
    status:"Scheduled"
  },
  {
    department:"VLSI",
    year:"Second Year",
    division:"A",
    room:"205",
    time:"09:00 - 10:00",
    subject:"Digital Electronics",
    teacher:"Professor C",
    status:"Scheduled"
  },
  {
    department:"Computer Engineering",
    year:"First Year",
    division:"A",
    room:"301",
    time:"09:00 - 10:00",
    subject:"Programming",
    teacher:"Professor D",
    status:"Scheduled"
  }
];

let swaps = [];
let notifications = [];

let role = "Principal";
let selectedLecture = null;

function notify(text) {
  notifications.unshift(text);
  document.getElementById("notificationList").innerHTML =
    notifications.map(n => <div class="notice">${n}</div>).join("");
}

function message(text) {
  const box = document.getElementById("message");
  box.textContent = text;
  box.style.display = "block";
  setTimeout(() => box.style.display = "none", 2500);
}

function changeRole() {
  role = document.getElementById("role").value;

  document.getElementById("welcomeText").textContent =
    "Logged in as " + role + ". Manage your college activities.";

  document.getElementById("requestSection").classList.toggle(
    "hidden", false
  );

  render();
  message("Role changed to " + role);
}

function showPage(page, element) {
  document.querySelectorAll(".page").forEach(p =>
    p.classList.add("hidden")
  );

  document.getElementById(page).classList.remove("hidden");

  document.querySelectorAll(".nav").forEach(n =>
    n.classList.remove("active")
  );

  element.classList.add("active");

  if(page === "classrooms") renderRooms();
  if(page === "timetable") renderTimetable();
  if(page === "swaps") renderSwaps();
}

function render() {
  const dept = document.getElementById("department").value;
  const year = document.getElementById("year").value;
  const div = document.getElementById("division").value;

  document.getElementById("selection").textContent =
    Selected: ${dept} / ${year} / Division ${div};

  const filtered = lectures.filter(l =>
    l.department === dept &&
    l.year === year &&
    l.division === div
  );

  document.getElementById("liveCount").textContent =
    lectures.filter(l => l.status === "Live").length;

  document.getElementById("swapCount").textContent =
    swaps.filter(s => s.status === "Pending").length;

  const box = document.getElementById("currentLecture");

  if(filtered.length === 0) {
    box.innerHTML = '<p class="muted">No sample timetable available for this selection.</p>';
    return;
  }

  box.innerHTML = filtered.map(l => `
    <div class="notice">
      <strong>${l.subject}</strong><br><br>
      Teacher: ${l.teacher}<br>
      Room: ${l.room}<br>
      Time: ${l.time}<br><br>
      <span class="status ${l.status === "Live" ? "live" : ""}">
        ${l.status}
      </span>
      <br><br>
      ${role !== "Principal" ? `
        <button class="action"
          onclick="startLecture(${lectures.indexOf(l)})"
          ${l.status === "Live" || l.status === "Completed" ? "disabled" : ""}>
          Start Lecture
        </button>
        <button class="action secondary"
          onclick="endLecture(${lectures.indexOf(l)})"
          ${l.status !== "Live" ? "disabled" : ""}>
          End Lecture
        </button>
      ` : ""}
    </div>
  `).join("");
}

function startLecture(index) {
  const lecture = lectures[index];

  if(lecture.status === "Live" ||
     lecture.status === "Completed") return;

  lecture.status = "Live";
  lecture.startedAt = new Date().toLocaleTimeString();

  notify(${lecture.teacher} started ${lecture.subject} in Room ${lecture.room}.);
  message("Lecture started successfully!");
  render();
  renderRooms();
}

function endLecture(index) {
  const lecture = lectures[index];

  if(lecture.status !== "Live") return;

  lecture.status = "Completed";
  lecture.endedAt = new Date().toLocaleTimeString();

  notify(${lecture.subject} in Room ${lecture.room} has ended.);
  message("Lecture ended!");
  render();
  renderRooms();
}

function renderRooms() {
  const dept = document.getElementById("roomDept").value;

  const filtered = lectures.filter(l =>
    l.department === dept
  );

  document.getElementById("rooms").innerHTML =
    filtered.length ? filtered.map(l => `
      <div class="notice">
        <strong>Room ${l.room}</strong> — ${l.division}<br><br>
        ${l.subject}<br>
        Teacher: ${l.teacher}<br>
        Time: ${l.time}<br><br>
        <span class="status ${l.status === "Live" ? "live" : ""}">
          ${l.status}
        </span>
      </div>
    `).join("") :
    '<p class="muted">No classroom records available.</p>';
}

function renderTimetable() {
  const rows = [
    ["09:00 - 10:00","Mathematics","Electronics","Physics","Mathematics","Programming"],
    ["10:00 - 11:00","Electronics","Mathematics","Programming","Physics","Electronics"],
    ["11:00 - 12:00","Physics","Programming","Mathematics","Electronics","Physics"],
    ["12:00 - 01:00","Lunch Break","Lunch Break","Lunch Break","Lunch Break","Lunch Break"],
    ["01:00 - 02:00","Programming","Physics","Electronics","Programming","Mathematics"]
  ];

  document.getElementById("timetableBody").innerHTML =
    rows.map(row => `
      <tr>${row.map(cell => <td>${cell}</td>).join("")}</tr>
    `).join("");
}

function requestSwap() {
  const from = document.getElementById("fromTeacher").value;
  const to = document.getElementById("toTeacher").value;
  const division = document.getElementById("swapDivision").value;

  if(from === to) {
    message("Choose two different teachers.");
    return;
  }

  swaps.push({
    from,
    to,
    division,
    status:"Pending"
  });

  notify(${from} requested a lecture swap with ${to} for ${division}.);

  renderSwaps();
  render();
  message("Swap request submitted!");
}

function renderSwaps() {
  const box = document.getElementById("swapRequests");

  if(swaps.length === 0) {
    box.innerHTML = '<p class="muted">No swap requests yet.</p>';
    return;
  }

  box.innerHTML = swaps.map((s,i) => `
    <div class="notice">
      <strong>${s.from}</strong> ⇄ <strong>${s.to}</strong><br><br>
      Division: ${s.division}<br>
      Status: <span class="status ${s.status === "Approved" ? "live" : "pending"}">${s.status}</span>
      <br><br>
      ${role === "HOD" || role === "Principal" ?
        s.status === "Pending" ?
        `<button class="action" onclick="approveSwap(${i})">Approve</button>
         <button class="action secondary" onclick="rejectSwap(${i})">Reject</button>`
        : "" : ""}
    </div>
  `).join("");
}

function approveSwap(index) {
  swaps[index].status = "Approved";
  notify(Swap between ${swaps[index].from} and ${swaps[index].to} approved.);
  renderSwaps();
  render();
  message("Swap approved!");
}

function rejectSwap(index) {
  swaps[index].status = "Rejected";
  notify(Swap request between ${swaps[index].from} and ${swaps[index].to} rejected.);
  renderSwaps();
  render();
  message("Swap rejected.");
}

// INITIALIZE

render();
renderRooms();
renderTimetable();
renderSwaps();

</script>
